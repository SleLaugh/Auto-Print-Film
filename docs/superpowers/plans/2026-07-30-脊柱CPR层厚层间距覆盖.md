# 脊柱 CPR 层厚层间距覆盖 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在现有脊柱冠状位 CPR 中叠加层厚、层间距和原始标注体素覆盖计算，保证 offset 序列不少包配置范围内的原始椎体标注。

**Implementation Status:** 已在当前工作区按本计划完成实现和验证；最终验证命令为 `.venv/bin/python -m pytest -q`。

**Architecture:** 阶段四仍由过滤椎体主体标注生成中心曲线，新增原始分割体素覆盖 helper 计算前后 offset 区间和必要张数。阶段五把 `CurvedStackResampler` 从单曲面采样扩展为厚层 slab 平均采样，继续消费现有 `SlicePlan` 字段，不新增阶段间协议字段。

**Tech Stack:** Python 3、NumPy、SimpleITK、unittest/pytest、现有 `SlicePlan`/`PlannedSlice` 数据模型。

## Global Constraints

- 中线弧度保持不变，不重新拟合、不按原始全量标注重算质心。
- 首尾长度可以沿现有首尾切线方向延伸，保证覆盖原始标注的头足外缘。
- 偏移方向使用现有 CPR 的 `offset_axis`，胸椎冠状位 CPR 即患者前后方向。
- 张数可以不对称，前侧和后侧分别按原始体素外缘计算。
- 中心层 `offset=0` 必须保留。
- 所有覆盖判定基于当前部位配置内的原始标签体素；颈椎 CPR 仍排除 C1，胸椎默认 C7 至 L1 全部参与。
- 不能因为延伸或覆盖计算把配置范围外的上下相邻椎体纳入 CPR 曲线或张数边界。
- 宁可最外层向外多覆盖几毫米，也不能少包任何配置范围内的原始体素。
- 第一版厚层 CPR 只实现算术平均融合，不加 MIP/MinIP 配置。

```python
def _raw_label_offset_interval_mm(
    localization_result: Dict[str, Any],
    label_names: List[str],
    curve_points: List[Point3D],
    offset_axis: Point3D,
) -> Tuple[float, float] | None:
    """计算原始标签体素相对现有 CPR 中线的前后偏移区间。"""

    segmentation = _read_original_segmentation(localization_result)
    if segmentation is None:
        return None
    label_ids = _configured_label_ids(localization_result, label_names)
    if not label_ids:
        return None

    curve = np.asarray([[point.x, point.y, point.z] for point in curve_points], dtype=float)
    if len(curve) < 2:
        return None
    axis = np.asarray([offset_axis.x, offset_axis.y, offset_axis.z], dtype=float)
    axis_length = float(np.linalg.norm(axis))
    if axis_length <= 1e-6 or not math.isfinite(axis_length):
        return None
    axis = axis / axis_length

    mask_points = _raw_label_physical_points(segmentation, label_ids)
    if mask_points.size == 0:
        return None
    curve_without_axis = curve - np.outer(curve @ axis, axis)
    points_without_axis = mask_points - np.outer(mask_points @ axis, axis)
    nearest_indices = _nearest_curve_indices(points_without_axis, curve_without_axis)
    offsets = np.sum((mask_points - curve[nearest_indices]) * axis, axis=1)
    if offsets.size == 0:
        return None

    safety_margin = _voxel_safety_margin_mm(segmentation)
    return (
        float(np.min(offsets) - safety_margin),
        float(np.max(offsets) + safety_margin),
    )


def _configured_label_ids(
    localization_result: Dict[str, Any],
    label_names: List[str],
) -> set[int]:
    """收集当前 CPR 配置范围内真实存在的原始分割 label id。"""

    label_ids: set[int] = set()
    coords = localization_result.get("coords", {})
    for label_name in label_names:
        values = coords.get(label_name, {}).get("label_ids")
        if not isinstance(values, (list, tuple, set)):
            continue
        for value in values:
            try:
                label_ids.add(int(value))
            except (TypeError, ValueError):
                continue
    return label_ids


def _raw_label_physical_points(
    segmentation: sitk.Image,
    label_ids: set[int],
) -> np.ndarray:
    """返回原始标签体素中心的物理坐标数组。"""

    array = sitk.GetArrayViewFromImage(segmentation)
    mask = np.isin(array, list(label_ids))
    indices_zyx = np.argwhere(mask)
    if indices_zyx.size == 0:
        return np.empty((0, 3), dtype=float)
    indices_xyz = indices_zyx[:, [2, 1, 0]].astype(float)
    origin = np.asarray(segmentation.GetOrigin(), dtype=float)
    spacing = np.asarray(segmentation.GetSpacing(), dtype=float)
    direction = np.asarray(segmentation.GetDirection(), dtype=float).reshape(3, 3)
    return origin + (indices_xyz * spacing) @ direction.T


def _nearest_curve_indices(points: np.ndarray, curve: np.ndarray) -> np.ndarray:
    """分块查找每个原始体素点最近的 CPR 曲线采样点。"""

    nearest = np.empty((len(points),), dtype=int)
    chunk_size = 4096
    for start in range(0, len(points), chunk_size):
        stop = min(start + chunk_size, len(points))
        distances = np.sum(
            (points[start:stop, np.newaxis, :] - curve[np.newaxis, :, :]) ** 2,
            axis=2,
        )
        nearest[start:stop] = np.argmin(distances, axis=1)
    return nearest


def _voxel_safety_margin_mm(image: sitk.Image) -> float:
    """用半个体素对角线作为抽样/体素中心到体素外缘的安全余量。"""

    spacing = [abs(float(value)) for value in image.GetSpacing()]
    return 0.5 * math.sqrt(sum(value ** 2 for value in spacing))
```

```python
def _offsets_covering_interval(
    d_min: float,
    d_max: float,
    slice_thickness: float,
    slice_spacing: float,
) -> List[float]:
    """生成不少包 `[d_min, d_max]` 的 CPR 厚层中心 offset。"""

    thickness = float(slice_thickness)
    spacing = float(slice_spacing)
    if thickness <= 0.0:
        raise ValueError("slice_thickness 必须大于 0")
    if spacing <= 0.0:
        raise ValueError("slice_spacing 必须大于 0")

    half_thickness = thickness / 2.0
    neg_count = int(math.ceil(max(0.0, -float(d_min) - half_thickness) / spacing))
    pos_count = int(math.ceil(max(0.0, float(d_max) - half_thickness) / spacing))
    while -neg_count * spacing - half_thickness > float(d_min) + 1e-6:
        neg_count += 1
    while pos_count * spacing + half_thickness < float(d_max) - 1e-6:
        pos_count += 1
    return [
        round(index * spacing, 6)
        for index in range(-neg_count, pos_count + 1)
    ]


def _bbox_offsets_mm(
    localization_result: Dict[str, Any],
    label_names: List[str],
    curve_points: List[Point3D],
    offset_axis: Point3D,
    slice_thickness: float,
    slice_spacing: float,
) -> List[float]:
    """原始分割不可用时，用完整 bbox 角点作为保守回退。"""

    center_projection = float(np.median([
        _dot(point, offset_axis)
        for point in curve_points
    ]))
    projections: list[float] = []
    for label_name in label_names:
        bbox = (
            localization_result
            .get("coords", {})
            .get(label_name, {})
            .get("bbox")
        )
        if not bbox:
            continue
        projections.extend(
            _dot(point, offset_axis) - center_projection
            for point in _bbox_corners(bbox, localization_result["image"])
        )

    if not projections:
        spacing = max(float(slice_spacing), 1e-6)
        projections = [-spacing, 0.0, spacing]
    return _offsets_covering_interval(
        min(projections),
        max(projections),
        slice_thickness,
        slice_spacing,
    )
```

```python
    def _sample_curved_surface_slab(
        self,
        image: sitk.Image,
        source: np.ndarray,
        curve_samples: np.ndarray,
        in_plane_directions: np.ndarray,
        offset_axis: np.ndarray,
        base_offset_mm: float,
        in_plane_offsets: np.ndarray,
        slab_offsets: list[float],
    ) -> np.ndarray:
        """采样一个 CPR 厚层 slab，并对多个子曲面做算术平均。"""

        rows = len(curve_samples)
        columns = len(in_plane_offsets)
        accumulated = np.zeros((rows, columns), dtype=np.float32)
        for slab_offset in slab_offsets:
            offset_mm = base_offset_mm + slab_offset
            for row in range(rows):
                in_plane_dir = in_plane_directions[row]
                row_center = curve_samples[row] + offset_axis * offset_mm
                for column, in_plane_offset in enumerate(in_plane_offsets):
                    point = row_center + in_plane_dir * in_plane_offset
                    accumulated[row, column] += self._sample_linear(image, source, point)
        return accumulated / float(len(slab_offsets))

    @staticmethod
    def _slab_offsets(image: sitk.Image, thickness: float) -> list[float]:
        """按输入最小体素间距在 CPR 层厚内生成居中的子曲面偏移。"""

        if thickness <= 0.0:
            raise ValueError("slice_thickness 必须大于 0")
        source_spacing = min(float(value) for value in image.GetSpacing())
        sample_step = min(source_spacing / 2.0, 0.25)
        sample_count = max(1, int(math.ceil(thickness / sample_step)))
        sample_step = thickness / sample_count
        return [
            -thickness / 2.0 + sample_step / 2.0 + index * sample_step
            for index in range(sample_count)
        ]
```
